typedef struct {
    // STATO
    float rpm;
    float v;        // Trend (RPM al secondo)

    // COVARIANZA
    float p11, p12;
    float p21, p22;

    // TARATURA
    float Q_rpm;    // Rumore processo RPM
    float Q_v;      // Rumore processo Accelerazione
    float R;        // Rumore Sensore
} KalmanFilter;

void kalman_init(KalmanFilter *kf) {
    kf->rpm = 0;
    kf->v = 0;

    // Incertezza iniziale
    kf->p11 = 1.0f; kf->p12 = 0.0f;
    kf->p21 = 0.0f; kf->p22 = 1.0f;

    // TARATURA (TUNING)
    // Nota: Q_v è la varianza dell'accelerazione.
    // Poiché il dt cambia, il filtro si adatta automaticamente.
    kf->Q_rpm = 0.1f;
    kf->Q_v   = 5.0f; // Aumenta se vuoi inseguire accelerazioni brucianti
    kf->R     = 10.0f;
}

// ORA LA FUNZIONE ACCETTA "dt" (in secondi)
void kalman_update(KalmanFilter *kf, float measured_rpm, float dt) {

    // SAFETY CHECK: Evita divisioni per zero o dt negativi/nulli
    if (dt <= 0.0001f) dt = 0.0001f;

    // 1. PREDIZIONE (Time Update) con dt dinamico
    float rpm_pred = kf->rpm + kf->v * dt;
    float v_pred   = kf->v;

    // Aggiornamento Covarianza P = F*P*F' + Q
    // F = [1  dt]
    //     [0   1]
    float p11_pred = kf->p11 + dt * (kf->p21 + kf->p12) + dt * dt * kf->p22 + kf->Q_rpm;
    float p12_pred = kf->p12 + dt * kf->p22;
    float p21_pred = kf->p21 + dt * kf->p22;
    float p22_pred = kf->p22 + kf->Q_v; // Qui si assume che l'errore di acc. cresca col tempo

    // 2. CORREZIONE (Measurement Update)
    float y = measured_rpm - rpm_pred;

    float S = p11_pred + kf->R;
    float K_rpm = p11_pred / S;
    float K_v   = p21_pred / S;

    // Aggiornamento Stato
    kf->rpm = rpm_pred + K_rpm * y;
    kf->v   = v_pred   + K_v   * y;

    // Aggiornamento Covarianza P = (I - K*H)*P
    // H = [1 0]
    kf->p11 = (1.0f - K_rpm) * p11_pred;
    kf->p12 = (1.0f - K_rpm) * p12_pred;
    kf->p21 = -K_v * p11_pred + p21_pred;
    kf->p22 = -K_v * p12_pred + p22_pred;
}

// Predizione futura basata sull'ultimo dt conosciuto
float kalman_predict_future(KalmanFilter *kf, float last_dt, int steps) {
    // Assumiamo che nei prossimi 'steps' giri, il dt rimanga simile all'ultimo misurato
    float total_time_ahead = last_dt * steps;
    return kf->rpm + (kf->v * total_time_ahead);
}

void update_rpm_R(){
	calc_timing_advance();

	TIM1->CR1 &= ~TIM_CR1_CEN;
	TIM1->PSC = 99;
	TIM1->CCMR1 &= ~TIM_CCMR1_OC1M; // Pulisci i bit precedenti per CH1
	TIM1->CCMR1 |= (TIM_CCMR1_OC1M_0 | TIM_CCMR1_OC1M_1 | TIM_CCMR1_OC1M_2);
	TIM1->CCMR1 |= TIM_CCMR1_OC1PE;
	TIM1->ARR = advance_time + t_on;  // Periodo Totale: 3 sec
	TIM1->CCR1 = advance_time; // Punto di scatto: 1 sec (Nota: CCR1 invece di CCR3)
	TIM1->CCER |= TIM_CCER_CC1E;
	TIM1->BDTR |= TIM_BDTR_MOE;
	TIM1->CR1 |= TIM_CR1_OPM;
	TIM1->EGR |= TIM_EGR_UG;
	TIM1->SR = 0;
	TIM1->CNT = 0;
	TIM1->CR1 |= TIM_CR1_CEN;
}

}

void calc_timing_advance(void) {

    // --- 1. ACQUISIZIONE TEMPO E RPM ---
    uint32_t current_time = TIM5->CNT;
    uint32_t delta_micros = current_time - last_interrupt_time;
    last_interrupt_time = current_time;

    // Filtro anti-rumore hardware (Debounce)
    if (delta_micros < 100) return;

    // Calcolo variabili base
    float dt_seconds = (float)delta_micros / 1000000.0f;
    float raw_rpm = 60000000.0f / delta_micros;

    // --- 2. FILTRO KALMAN E PREDIZIONE ---
    kalman_update(&kf, raw_rpm, dt_seconds);

    // Prediciamo i giri futuri per compensare l'accelerazione
    float next_rpm = kalman_predict_future(&kf, dt_seconds, 1);

    // Clamp RPM per evitare divisioni per zero o valori assurdi
    if (next_rpm < 100.0f) next_rpm = 100.0f;
    if (next_rpm > 18000.0f) next_rpm = 18000.0f;

    // --- 3. LETTURA MAPPA ANTICIPO ---
    float target_advance = 0.0f;

    if (next_rpm <= map_rpm_axis[0]) {
        target_advance = map_advance_curve[0];
    }
    else if (next_rpm >= map_rpm_axis[IGN_MAP_ROWS - 1]) {
        target_advance = map_advance_curve[IGN_MAP_ROWS - 1];
    }
    else {
        // Interpolazione lineare
        int i = 0;
        while (map_rpm_axis[i + 1] < next_rpm) {
            i++;
        }
        float x1 = (float)map_rpm_axis[i];
        float x2 = (float)map_rpm_axis[i + 1];
        float y1 = (float)map_advance_curve[i];
        float y2 = (float)map_advance_curve[i + 1];

        float ratio = (next_rpm - x1) / (x2 - x1);
        target_advance = y1 + (y2 - y1) * ratio;
    }

    // --- 4. SAFETY GUARDS (SALVA-MOTORE) ---

    // A. PROTEZIONE ANTI-CONTRACCOLPO IN AVVIAMENTO
    // Se i giri sono bassi (avviamento), ignora la mappa e usa un anticipo fisso e sicuro.
    // Esempio: 10 gradi. Se provi ad accendere a 30° mentre pedali/usi il motorino, spacchi tutto.
    if (next_rpm < 600.0f) {
        target_advance = 10.0f;
    }

    // B. PROTEZIONE FISICA (PICKUP LIMIT)
    // Non possiamo dare più anticipo di dove si trova il pickup fisicamente.
    // Lasciamo un margine di sicurezza di 2 gradi per il tempo di elaborazione.
    float max_possible_advance = FIXED_PULSER_ADVANCE - 2.0f;

    if (target_advance > max_possible_advance) {
        target_advance = max_possible_advance;
    }

    // --- 5. CALCOLO DEL TEMPO DI RITARDO ---

    // Calcoliamo quanti gradi dobbiamo ASPETTARE prima di scoccare
    float degrees_to_wait = FIXED_PULSER_ADVANCE - target_advance;

    // C. PROTEZIONE MATEMATICA NEGATIVA
    // Se per qualche errore di arrotondamento viene < 0, forziamo un minimo
    if (degrees_to_wait < 0.5f) {
        degrees_to_wait = 0.5f;
    }

    // Calcolo finale del tempo in microsecondi (Tick del timer)
    // Formula: (Gradi * 166666.67) / RPM
    uint32_t calc_time = (uint32_t)((degrees_to_wait * 166666.67f) / next_rpm);

    // D. PROTEZIONE HARDWARE TIMER (MINIMUM DELAY)
    // Il timer non può gestire 0 o 1 microsecondo di ritardo in modo affidabile
    // (serve tempo per caricare i registri e lanciare l'ISR).
    // Forziamo un minimo di 50 microsecondi (o tick).
    if (calc_time < 50) {
        calc_time = 50;
    }

    // Salviamo nelle variabili globali VOLATILE
    advance_degree = (uint16_t)target_advance;
    advance_time = calc_time;
}
