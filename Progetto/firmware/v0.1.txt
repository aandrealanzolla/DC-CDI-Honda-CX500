void update_rpm_R(){
	//HAL_TIM_PWM_Stop_IT(&htim1,TIM_CHANNEL_1);7
//	HAL_TIM_OnePulse_Stop_IT(&htim1, TIM_CHANNEL_1);
	if(right_cyl_trig==1) {
		t1_r = timerCounter;
		right_cyl_trig = 0;
	}
	// RISOLUZIONE MAX TEMPO = 100us (deltaT_min)
	deltaT_r = (float) (t1_r-t0_r)*0.1f; //converto la differenza da centinaia di us in ms
	if(deltaT_r < 5000.0f ) {
		RPM_R = 60000.0f/((float)(deltaT_r)); // the time difference used for the rpm estimation is the mean between the delta_T on the right and left cylinder
		calc_timing_advance(RPM);
		TIM1->CR1 &= ~TIM_CR1_CEN;

		TIM1->CCMR1 &= ~TIM_CCMR1_OC1M;
		TIM1->CCMR1 |= (TIM_CCMR1_OC1M_0 | TIM_CCMR1_OC1M_1 | TIM_CCMR1_OC1M_2);
		TIM1->CCER |= TIM_CCER_CC1E;

		TIM1->ARR = (int) (advance_time) + t_on;
		TIM1->CCR1 = (int) (advance_time);

		TIM1->CR1 |= TIM_CR1_OPM;
		TIM1->EGR |= TIM_EGR_UG;
		TIM1->SR = 0;
		TIM1->CNT = 0;
		TIM1->BDTR |= TIM_BDTR_MOE;
		TIM1->CR1 |= TIM_CR1_CEN;
		t0_r = t1_r;
	}
}


void update_rpm_L(){
	if(left_cyl_trig==1) {
		left_cyl_trig = 0;
		t1_l = timerCounter;
	}
	// RISOLUZIONE MAX TEMPO = 100us (deltaT_min)
	deltaT_l = (float) (t1_l-t0_l)*0.1f;
	if(deltaT_l < 5000.0f) {
		RPM_L = 60000.0f/((float)(deltaT_l)); // the time difference used for the rpm estimation is the mean between the delta_T on the right and left cylinder

		calc_timing_advance(RPM);
		TIM2->CR1 &= ~TIM_CR1_CEN;

		TIM2->CCMR1 &= ~TIM_CCMR1_OC1M;
		TIM2->CCMR1 |= (TIM_CCMR1_OC1M_0 | TIM_CCMR1_OC1M_1 | TIM_CCMR1_OC1M_2);
		TIM2->CCER |= TIM_CCER_CC1E;

		TIM2->ARR = (int) (advance_time) + t_on;
		TIM2->CCR1 = (int) (advance_time);

		TIM2->CR1 |= TIM_CR1_OPM;
		TIM2->EGR |= TIM_EGR_UG;
		TIM2->SR = 0;
		TIM2->CNT = 0;
		//TIM1->BDTR |= TIM_BDTR_MOE;
		TIM2->CR1 |= TIM_CR1_CEN;
		t0_l = t1_l;
	}
}


void calc_timing_advance(uint16_t rpm) {

	//Gestione Casi Limite (Fuori Mappa)
	if (rpm <= map_rpm_axis[0]) return map_advance_curve[0];
	if (rpm >= map_rpm_axis[IGN_MAP_ROWS - 1]) return map_advance_curve[IGN_MAP_ROWS - 1];

	//trova l'intervallo
	int i = 0;
	while (map_rpm_axis[i + 1] < rpm) {
		i++;
	}
	// interpolazione
	uint16_t x1 = map_rpm_axis[i];
	uint16_t x2 = map_rpm_axis[i + 1];
	uint8_t  y1 = map_advance_curve[i];
	uint8_t  y2 = map_advance_curve[i + 1];

	float ratio = (float)(rpm - x1) / (float)(x2 - x1);

	advance_degree = y1 + (y2 - y1) * ratio;
	//advance_degree = 15;

	//if(advance_degree <= 15){
		advance_time = ((FIXED_PULSER_ADVANCE-advance_degree) * 1000000.0f) / (RPM * 6.0f);
	//}
	/*
	else{
		advance_time = ((380-advance_degree) * 1000000.0f) / (RPM * 6.0f);
	}
	//advance_time = ((FIXED_PULSER_ADVANCE-advance_degree) * 1000000.0f) / (RPM * 6.0f);
*/
}

void engine_off_check(){

	if((timerCounter - t1_l > 1000) || (timerCounter - t1_r > 1000)){
		RPM_R = 0.0f;
		RPM_L = 0.0f;
//		advance_degree = FIXED_PULSER_ADVANCE - 5;
		//HAL_TIM_PWM_Stop_IT(&htim1,TIM_CHANNEL_1);
		HAL_TIM_OC_Stop(&htim1, TIM_CHANNEL_1);
		HAL_TIM_OC_Stop(&htim2, TIM_CHANNEL_1);
	}
}
